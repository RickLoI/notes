    \begin{figure}
        \begin{tikzpicture}[scale=0.9]
            % --- First elliptic curve: y^2 = x^3 - x + 1 (one connected component) ---
            \begin{scope}
                \draw[->] (-1.5,0) -- (2.5,0) node[right] {$x$};
                \draw[->] (0,-2.5) -- (0,2.5) node[above] {$y$};
                \draw[domain=-1.3247:2.1, samples=150, smooth, variable=\x, thick, hColumbine]
                    plot ({\x}, {sqrt(max(\x*\x*\x - \x + 1, 0))});
                \draw[domain=-1.3247:2.1, samples=150, smooth, variable=\x, thick, hColumbine]
                    plot ({\x}, {-sqrt(max(\x*\x*\x - \x + 1, 0))});
            \end{scope}

            % --- Second elliptic curve: y^2 = x^3 - x (two components) ---
            \begin{scope}[xshift=7cm]
                \draw[->] (-1.5,0) -- (2.5,0) node[right] {$x$};
                \draw[->] (0,-2.5) -- (0,2.5) node[above] {$y$};
                % Oval component, x in [-1,0]
                \draw[domain=-1:0, samples=80, smooth, variable=\x, thick, hAzalea]
                    plot ({\x}, {sqrt(max(\x*\x*\x - \x, 0))});
                \draw[domain=-1:0, samples=80, smooth, variable=\x, thick, hAzalea]
                    plot ({\x}, {-sqrt(max(\x*\x*\x - \x, 0))});
                % Unbounded component, x in [1,2]
                \draw[domain=1:2, samples=80, smooth, variable=\x, thick, hAzalea]
                    plot ({\x}, {sqrt(max(\x*\x*\x - \x, 0))});
                \draw[domain=1:2, samples=80, smooth, variable=\x, thick, hAzalea]
                    plot ({\x}, {-sqrt(max(\x*\x*\x - \x, 0))});
            \end{scope}
        \end{tikzpicture} 
        \caption{Example of elliptic curves over \(\rset\).}
        \label{fig:elliptic-curves-over-R}
    \end{figure}
\end{document}
